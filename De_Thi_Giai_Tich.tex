    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 0} 2 x^{2} - 4 x + 6 $

    A. $ -6 $ \qquad B. $ 6 $ \qquad C. $ 1 $ \qquad D. $ 7 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 1} 3 x^{2} - 2 x + 4 $

    A. $ 5 $ \qquad B. $ -\infty $ \qquad C. $ \infty $ \qquad D. $ 4 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 0} 3 x^{2} + x + 6 $

    A. $ 5 $ \qquad B. $ 0 $ \qquad C. $ \text{Không tồn tại} $ \qquad D. $ 6 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 0} 3 x^{2} + x + 10 $

    A. $ -\infty $ \qquad B. $ 0 $ \qquad C. $ 10 $ \qquad D. $ -1 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to \infty} \frac{5 x^{2} + 1}{3 x^{2} + 2} $

    A. $ \infty $ \qquad B. $ \frac{5}{3} $ \qquad C. $ - \frac{5}{3} $ \qquad D. $ -1 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 3} \frac{5 x^{2} - 45}{x - 3} $

    A. $ 0 $ \qquad B. $ -1 $ \qquad C. $ 31 $ \qquad D. $ 30 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to \infty} \frac{5 x^{2} + 1}{4 x^{2} + 2} $

    A. $ \text{Không tồn tại} $ \qquad B. $ \infty $ \qquad C. $ \frac{5}{4} $ \qquad D. $ \frac{1}{4} $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 0} \frac{e^{3 x} - 1}{x} $

    A. $ 1 $ \qquad B. $ 3 $ \qquad C. $ \infty $ \qquad D. $ -3 $

    \vspace{0.3cm}

    \item Tính giới hạn sau: $ \displaystyle \lim_{x \to 0} \frac{e^{3 x} - 1}{x} $

    A. $ 3 $ \qquad B. $ \infty $ \qquad C. $ \text{Không tồn tại} $ \qquad D. $ -3 $

    \vspace{0.3cm}

\end{enumerate}
